先把 `cleanupChildren` 补齐：复制当前 `owner.children` 快照并清空集合，再按后进先出顺序逐个调用 `disposeOwner` 递归释放所有后代：

```ts
function cleanupChildren(owner: Owner): void {
  // 先复制快照并清空 owner.children，再按逆序逐个调用 disposeOwner 递归清理后代。
  const children = Array.from(owner.children);
  owner.children.clear();
  for (const child of children.reverse()) {
    disposeOwner(child);
  }
}
```

接着在 `runEffect` 中补上重跑前的 `cleanupChildren(effect)` 以及 `currentOwner` 的成对切换与恢复，并在 `createEffect` 里把新创建的节点登记到当前活跃的 `currentOwner.children`：

```ts
function runEffect(effect: Effect): void {
  if (effect.disposed) return;
  cleanupDependencies(effect);
  cleanupChildren(effect);

  const previousEffect = currentEffect;
  const previousOwner = currentOwner;
  currentEffect = effect;
  currentOwner = effect;
  try {
    effect.callback();
  } catch (error) {
    disposeOwner(effect);
    throw error;
  } finally {
    currentEffect = previousEffect;
    currentOwner = previousOwner;
  }
}
```

```ts
export function createEffect(callback: EffectCallback): Dispose {
  const effect: Effect = {
    parent: currentOwner,
    children: new Set(),
    disposed: false,
    callback,
    dependencies: new Set(),
  };
  effect.onDispose = () => cleanupDependencies(effect);

  if (currentOwner && !currentOwner.disposed) {
    currentOwner.children.add(effect);
  }

  runEffect(effect);
  return () => disposeOwner(effect);
}
```
